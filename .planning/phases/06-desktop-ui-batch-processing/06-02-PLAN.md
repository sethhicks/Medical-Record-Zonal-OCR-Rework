**_worker(self)** method (runs on background thread — NEVER touches tkinter widgets):
```python
def _worker(self):
    import pipeline
    files = list(self._files)
    settings = self.settings
    # Count total pages across all files
    from pdf2image import pdfinfo_from_path
    total = 0
    page_counts = []
    for pdf_path in files:
        try:
            info = pdfinfo_from_path(pdf_path, poppler_path=settings.get("poppler_path"))
            n_pages = info["Pages"]
        except Exception:
            n_pages = 0
        page_counts.append(n_pages)
        total += n_pages

    cms_pages = []
    ub_pages = []
    current = 0

    for pdf_path, n_pages in zip(files, page_counts):
        for page_num in range(n_pages):
            current += 1
            self._queue.put(("progress", current, total))
            try:
                raw = pipeline.convert_page(pdf_path, page_num)
                form = pipeline.detect_form_type(raw)
                if form == "UNKNOWN":
                    self._queue.put(("error", pdf_path, page_num,
                                     "UNKNOWN form type"))
                else:
                    proc = pipeline.preprocess_page(raw, settings)
                    if form == "CMS-1500":
                        results = pipeline.extract_cms1500(proc, settings)
                        cms_pages.append(results)
                    else:
                        results = pipeline.extract_ub04(proc, settings)
                        ub_pages.append(results)
            except Exception as exc:
                self._queue.put(("error", pdf_path, page_num, str(exc)))

    # Write workbook — WR-02 fix: ensure output_dir exists
    try:
        os.makedirs(settings.get("output_dir", str(Path.home() / "Desktop")),
                    exist_ok=True)
        output_path = pipeline.write_workbook(cms_pages, ub_pages, settings)
    except Exception as exc:
        output_path = None
        self._queue.put(("error", "", 0, f"write_workbook failed: {exc}"))

    self._queue.put(("done", output_path))
```

**poll_queue(self)** method (called every 100ms via root.after — reads all queued items per tick):
```python
def poll_queue(self):
    done = False
    try:
        while True:
            msg = self._queue.get_nowait()
            kind = msg[0]
            if kind == "progress":
                _, current, total = msg
                pct = int(current / total * 100) if total > 0 else 0
                self.progress_bar["value"] = pct
                self.lbl_page.config(text=f"Page {current} / {total}")
            elif kind == "error":
                _, filename, page_num, error_message = msg
                basename = os.path.basename(filename) if filename else "unknown"
                line = f"{basename} page {page_num}: {error_message}\n"
                self.error_log.config(state=tk.NORMAL)
                self.error_log.insert(tk.END, line)
                self.error_log.see(tk.END)
                self.error_log.config(state=tk.DISABLED)
            elif kind == "done":
                _, output_path = msg
                self._output_path = output_path
                done = True
    except queue.Empty:
        pass

    if done:
        self._on_done()
    else:
        self.root.after(100, self.poll_queue)
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| user → filedialog | User selects PDF file paths via OS dialog; paths are untrusted filesystem locations |
| worker thread → main thread | queue.Queue bridges threads; widget updates only happen on main thread via poll_queue |
| main.py → os.startfile | Output path passed to os.startfile; must be a file that exists |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-06-02 | Tampering | filedialog path selection | accept | OS file dialog filters to *.pdf; user is the operator on their own machine; single-user desktop tool |
| T-06-03 | Denial of Service | worker thread infinite hang | mitigate | Worker thread is daemon=True; if main thread exits, worker is killed automatically; no explicit timeout needed for desktop tool |
| T-06-04 | Information Disclosure | error log widget | accept | Error messages contain filenames and page numbers — all data user already knows; no PII added |
| T-06-05 | Elevation of Privilege | os.startfile(output_path) | accept | Opens a file the worker just wrote; path is constructed from settings['output_dir'] + hardcoded filename; no user-controlled path injection |
| T-06-06 | Spoofing | run_checks before Tk() | mitigate | If run_checks() returns False, sys.exit(1) prevents window from opening; messagebox.showerror shown before exit |
</threat_model>
