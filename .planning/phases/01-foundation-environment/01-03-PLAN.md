```python
# config_loader.py
"""Settings loader for OCR Medical Billing Form Extractor.

Reads settings.json from the same directory as this file and merges user values
over hardcoded Windows defaults. Returns defaults when settings.json is absent.
Never raises on a missing file.
"""
import json
import pathlib

_DEFAULTS: dict = {
    "tesseract_cmd": r"C:\Program Files\Tesseract-OCR\tesseract.exe",
    "poppler_path": r"C:\Program Files\poppler\Library\bin",
    "confidence_threshold": 60,
    "output_dir": str(pathlib.Path.home() / "Desktop"),
}


def load_settings(path: str | None = None) -> dict:
    """Load settings.json and merge with built-in defaults.

    Args:
        path: Explicit path to settings.json. Defaults to settings.json in the
              same directory as this module (script-relative, not CWD-relative).

    Returns:
        dict with keys: tesseract_cmd, poppler_path, confidence_threshold, output_dir.
        User keys override defaults; missing keys fall back to defaults.

    Raises:
        ValueError: If settings.json exists but contains invalid JSON.
    """
    if path is None:
        path = str(pathlib.Path(__file__).parent / "settings.json")
    try:
        with open(path, "r", encoding="utf-8") as f:
            user = json.load(f)
        return {**_DEFAULTS, **user}   # user keys override defaults
    except FileNotFoundError:
        return _DEFAULTS.copy()
    except json.JSONDecodeError as e:
        raise ValueError(f"settings.json is not valid JSON: {e}") from e
```

Important implementation details:
- `path` parameter defaults to `None`, not `"settings.json"` — avoids CWD-relative resolution
- `Path(__file__).parent / "settings.json"` anchors to the directory containing config_loader.py
- `{**_DEFAULTS, **user}` is the merge — user keys win, no deep merge needed
- `FileNotFoundError` → return defaults (do NOT re-raise)
- `json.JSONDecodeError` → raise `ValueError` with the clear message shown above
- No module-level I/O — `_DEFAULTS` is a constant dict, `load_settings` is only called explicitly

<threat_model>
| Threat | STRIDE | Mitigation |
|--------|--------|-----------|
| Malicious settings.json with path injection in tesseract_cmd or poppler_path | Tampering | Paths from settings.json are passed to subprocess.run as list elements with shell=False (enforced in setup_check.py, Plan 01-04); no shell interpolation occurs; malicious path fails FileNotFoundError check cleanly |
| settings.json JSON parse bomb (deeply nested payload) | Tampering | stdlib json parser is resistant to deeply nested payloads; JSONDecodeError is caught and converted to ValueError with a user-facing message; no recursive deserialization |
| CWD-relative settings.json resolution allowing override from attacker-controlled directory | Tampering | Default path is anchored via Path(__file__).parent — attacker would need write access to the project directory itself, which is equivalent to arbitrary code execution |
</threat_model>
