```python
import sys
sys.path.insert(0, '.')
from config_loader import load_settings
from pipeline import convert_page, preprocess_page, detect_form_type, extract_cms1500, extract_ub04

settings = load_settings()
pdf = 'test.pdf'
cms_pages = []
ub04_pages = []

for page_num in range(30):
    try:
        raw = convert_page(pdf, page_num)
        form = detect_form_type(raw)
        proc = preprocess_page(raw, settings)
        if form == 'CMS-1500':
            results = extract_cms1500(proc, settings)
            non_empty = sum(1 for r in results if r.value)
            rate = non_empty / len(results)
            cms_pages.append((page_num, rate, non_empty, len(results)))
            print(f'Page {page_num:2d} CMS-1500: {non_empty:3d}/{len(results)} = {rate:.1%}')
        elif form == 'UB-04':
            results = extract_ub04(proc, settings)
            non_empty = sum(1 for r in results if r.value)
            rate = non_empty / len(results)
            ub04_pages.append((page_num, rate, non_empty, len(results)))
            print(f'Page {page_num:2d} UB-04:   {non_empty:3d}/{len(results)} = {rate:.1%}')
        else:
            print(f'Page {page_num:2d} UNKNOWN')
    except Exception as e:
        print(f'Page {page_num:2d} ERROR: {e}')

if cms_pages:
    best_cms = max(cms_pages, key=lambda x: x[1])
    print(f'\\nBest CMS-1500: page {best_cms[0]} at {best_cms[1]:.1%} ({best_cms[2]}/{best_cms[3]})')
if ub04_pages:
    best_ub04 = max(ub04_pages, key=lambda x: x[1])
    print(f'Best UB-04:   page {best_ub04[0]} at {best_ub04[1]:.1%} ({best_ub04[2]}/{best_ub04[3]})')
"
```

```python
def test_cms1500_whitelist_npi_chars(test_pdf_path, sample_settings):
    """CMS-1500 NPI and CPT fields contain only whitelisted characters (EXTR-01, D-02, D-03).

    Uses best CMS-1500 page from calibration sweep (page N — see STATE.md Phase 4 Calibration).
    """
    from pipeline import convert_page, preprocess_page, extract_cms1500

    raw = convert_page(test_pdf_path, BEST_CMS_PAGE)   # replace BEST_CMS_PAGE with actual int
    proc = preprocess_page(raw, sample_settings)
    results = extract_cms1500(proc, sample_settings)

    # NPI fields whitelist: digits only
    npi_whitelist = set("0123456789")
    npi_fields = [r for r in results if 'npi' in r.field_name and r.value]
    for r in npi_fields:
        invalid = set(r.value.replace(' ', '')) - npi_whitelist
        assert not invalid, f"{r.field_name}={r.value!r} contains invalid chars: {invalid}"

    # CPT field whitelist: digits, hyphen, space
    cpt_whitelist = set("0123456789- ")
    cpt_fields = [r for r in results if r.field_name.startswith('box24_cpt') and r.value]
    for r in cpt_fields:
        invalid = set(r.value) - cpt_whitelist
        assert not invalid, f"{r.field_name}={r.value!r} contains invalid chars: {invalid}"
```
